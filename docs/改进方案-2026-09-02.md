### P0-1：readJsonBody 增加 body 大小限制

**文件**：`plugins/dsh-permission-guard/lib/index.js`

**修改内容**：在 `readJsonBody` 函数中添加 1MB 大小限制，超过则拒绝请求。

**代码 diff 草案**：
```javascript
// 修改前
function readJsonBody(req) {
  return new Promise((resolve) => {
    const chunks = []
    req.on('data', (c) => chunks.push(c))
    req.on('end', () => {
      try { resolve(JSON.parse(Buffer.concat(chunks).toString('utf8') || '{}')) }
      catch { resolve({}) }
    })
    req.on('error', () => resolve({}))
  })
}

// 修改后
function readJsonBody(req) {
  const MAX_BODY_SIZE = 1024 * 1024 // 1MB limit
  return new Promise((resolve, reject) => {
    const chunks = []
    let totalSize = 0
    req.on('data', (c) => {
      totalSize += c.length
      if (totalSize > MAX_BODY_SIZE) {
        req.destroy() // Stop reading
        reject(new Error('Body too large'))
        return
      }
      chunks.push(c)
    })
    req.on('end', () => {
      try { resolve(JSON.parse(Buffer.concat(chunks).toString('utf8') || '{}')) }
      catch { resolve({}) }
    })
    req.on('error', (err) => {
      if (err.message === 'Body too large') {
        reject(err)
      } else {
        resolve({})
      }
    })
  })
}
```

### P1-3：open_workspace_path 路径校验

**文件**：`src-tauri/src/managers/window_ops.rs:104-118`

**修改内容**：添加路径校验，只允许打开 `$DSH_HOME` 或其子目录，拒绝任意路径。

**代码 diff 草案**：
```rust
// 修改前

    let path = if path.trim().is_empty() {
        crate::state::dsh_home().to_string_lossy().to_string()
    } else {
        path
    };
    log::info!("open_workspace_path invoked from page: {path}");

    let result = std::process::Command::new("explorer").arg(&path).spawn();
    // ...

// 修改后
pub fn open_workspace_path(path: String) -> Result<(), String> {
    let path = if path.trim().is_empty() {
        crate::state::dsh_home().to_string_lossy().to_string()
    } else {
        // Path validation: only allow $DSH_HOME and its subdirectories
        let dsh_home = crate::state::dsh_home();
        let target_path = std::path::Path::new(&path);
        
        // Resolve to absolute path and check containment
        if let Ok(abs_path) = std::fs::canonicalize(target_path) {
            if !abs_path.starts_with(&dsh_home) {
                log::warn!("open_workspace_path: path outside DSH_HOME rejected: {path}");
                return Err("Path outside DSH_HOME is not allowed".to_string());
            }
        } else {
            // Path doesn't exist or can't be resolved - allow for new directories
            // but log a warning
            log::info!("open_workspace_path: path not found, allowing for creation: {path}");
        }
        path
    };
    
    log::info!("open_workspace_path invoked from page: {path}");
    #[cfg(target_os = "windows")]
    let result = std::process::Command::new("explorer").arg(&path).spawn();
    #[cfg(target_os = "macos")]
    let result = std::process::Command::new("open").arg(&path).spawn();
    #[cfg(target_os = "linux")]
    let result = std::process::Command::new("xdg-open").arg(&path).spawn();
    result.map(|_| ()).map_err(|e| e.to_string())
}
```

### P2-6：Token HTML 注入转义

**文件**：`src/server/token.ts:38`

**修改内容**：对 `getToken()` 结果进行 HTML 转义，防止潜在 XSS。

**代码 diff 草案**：
```typescript
// 修改前
export function injectTokenToHtml(html: string): string {
  return html.replace(/<head[^>]*>/i, `<head><meta name="dsh-hub-token" content="${getToken()}">`)
}

// 修改后
function escapeHtml(str: string): string {
  return str
    .replace(/&/g, '&amp;')
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;')
    .replace(/'/g, '&#39;')
}

export function injectTokenToHtml(html: string): string {
  const token = escapeHtml(getToken())
  return html.replace(/<head[^>]*>/i, `<head><meta name="dsh-hub-token" content="${token}">`)
}
```
